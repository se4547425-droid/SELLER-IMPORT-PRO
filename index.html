document.addEventListener("DOMContentLoaded", () => { const API = "/api"; const pageTitles = { home: "Главная", request: "Рассчитать заказ", account: "Мои заявки", login: "Вход в аккаунт", register: "Регистрация" }; let token = localStorage.getItem("sip_token") || ""; async function api(path, options = {}) { const headers = { "Content-Type": "application/json", ...(token ? { Authorization: `Bearer ${token}` } : {}), ...options.headers }; const response = await fetch(API + path, { ...options, headers }); const data = await response.json(); if (!response.ok) { throw new Error(data.error || "Ошибка сервера"); } return data; } function showPage(name) { const target = document.getElementById(`page-${name}`); if (!target) return; document.querySelectorAll(".page").forEach(page => { page.classList.toggle("active", page === target); }); document.querySelectorAll("[data-page]").forEach(button => { button.classList.toggle("active", button.dataset.page === name); }); const title = document.getElementById("page-title"); if (title) title.textContent = pageTitles[name] || "SELLER IMPORT PRO"; window.scrollTo({ top: 0, behavior: "smooth" }); } function message(id, text, error = false) { const element = document.getElementById(id); if (!element) return; element.textContent = text; element.style.color = error ? "#ff7777" : "#67d9a2"; } function updateAccountButtons() { const loginButtons = document.querySelectorAll( '[data-page="login"]' ); loginButtons.forEach(button => { button.textContent = token ? "Выйти" : "Войти"; button.dataset.action = token ? "logout" : ""; }); } document.addEventListener("click", async event => { const button = event.target.closest("[data-page]"); if (!button) return; event.preventDefault(); if (button.dataset.action === "logout") { try { await api("/logout", { method: "POST" }); } catch (error) { console.error(error); } token = ""; localStorage.removeItem("sip_token"); updateAccountButtons(); showPage("home"); return; } const page = button.dataset.page; if (page === "account" && token) { await loadOrders(); } showPage(page); }); // Регистрация const registerForm = document.getElementById("register-form"); registerForm?.addEventListener("submit", async event => { event.preventDefault(); message("register-message", "Создаём аккаунт..."); try { const result = await api("/register", { method: "POST", body: JSON.stringify({ email: document.getElementById("register-email").value, password: document.getElementById("register-password").value }) }); token = result.token; localStorage.setItem("sip_token", token); updateAccountButtons(); message("register-message", "Аккаунт успешно создан!"); registerForm.reset(); showPage("account"); await loadOrders(); } catch (error) { message("register-message", error.message, true); } }); // Вход const loginForm = document.getElementById("login-form"); loginForm?.addEventListener("submit", async event => { event.preventDefault(); message("login-message", "Выполняется вход..."); try { const result = await api("/login", { method: "POST", body: JSON.stringify({ email: document.getElementById("login-email").value, password: document.getElementById("login-password").value }) }); token = result.token; localStorage.setItem("sip_token", token); updateAccountButtons(); message("login-message", "Вход выполнен!"); loginForm.reset(); showPage("account"); await loadOrders(); } catch (error) { message("login-message", error.message, true); } }); // Создание заявки const requestForm = document.getElementById("request-form"); requestForm?.addEventListener("submit", async event => { event.preventDefault(); if (!token) { message("request-message", "Сначала войдите или создайте аккаунт.", true); showPage("login"); return; } message("request-message", "Отправляем заявку..."); try { const result = await api("/orders", { method: "POST", body: JSON.stringify({ product_url: document.getElementById("product-url").value, product_price_cny: document.getElementById("product-price").value }) }); message("request-message", `Заявка №${result.order_id} успешно создана!`); requestForm.reset(); } catch (error) { message("request-message", error.message, true); } }); // Список заявок async function loadOrders() { const section = document.getElementById("page-account"); if (!section) return; if (!token) { section.innerHTML = `

Войдите в аккаунт

Чтобы увидеть свои заявки, необходимо авторизоваться.

`; return; } try { const result = await api("/orders"); if (!result.orders.length) { section.innerHTML = `
▤

Пока нет заявок

Создайте первую заявку на товар из Китая.

`; return; } const rows = result.orders.map(order => `
ЗАЯВКА №${order.id}

${escapeHtml(order.product_url)}

Цена товара: ${Number(order.product_price_cny).toFixed(2)} ¥

Вес: ${order.weight_kg == null ? "Уточняется" : Number(order.weight_kg) + " кг"}

Итоговая стоимость: ${ order.total_price_rub == null ? "Готовится" : Number(order.total_price_rub).toFixed(2) + " ₽" }

Статус: ${escapeHtml(order.status)}

`).join(""); section.innerHTML = `
ЛИЧНЫЙ КАБИНЕТ

Мои заявки

${rows}
`; } catch (error) { if (error.message === "Требуется вход") { token = ""; localStorage.removeItem("sip_token"); updateAccountButtons(); } section.innerHTML = `

Не удалось загрузить заявки

${escapeHtml(error.message)}

`; } } function escapeHtml(value) { return String(value).replace(/[&<>"']/g, character => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[character]); } updateAccountButtons(); if (token) { api("/me") .then(() => updateAccountButtons()) .catch(() => { token = ""; localStorage.removeItem("sip_token"); updateAccountButtons(); }); } console.log("SELLER IMPORT PRO: JavaScript загружен успешно"); });
